cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(grey-crow-speech-input LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
if(NOT DEFINED SHERPA_ONNX_SOURCE_DIR)
  message(FATAL_ERROR "Use scripts/prepare-speech-input-runtime.js to supply pinned sherpa sources")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature PYTHON TESTS PORTAUDIO JNI WEBSOCKET GPU DIRECTML BINARY TTS SPEAKER_DIARIZATION)
  set(SHERPA_ONNX_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_USE_PRE_INSTALLED_ONNXRUNTIME_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
# __FILE__ diagnostics must not disclose a developer's home/checkout path in
# shipped objects. Upstream precompiled ORT retains its public CI build paths.
foreach(kind CMAKE_SOURCE_DIR SHERPA_ONNX_SOURCE_DIR CMAKE_BINARY_DIR)
  get_filename_component(normalized "${${kind}}" REALPATH)
  # On macOS CMake can spell the same directory as /tmp and /private/tmp.
  string(REGEX REPLACE "^/private/" "/" mac_alias "${normalized}")
  if(kind STREQUAL "CMAKE_SOURCE_DIR")
    set(mapped "/grey-crow/speech-input")
  elseif(kind STREQUAL "SHERPA_ONNX_SOURCE_DIR")
    set(mapped "/third-party/sherpa-onnx")
  else()
    set(mapped "/third-party/build")
  endif()
  foreach(source_path "${${kind}}" "${normalized}" "${mac_alias}")
    if(MSVC)
      add_compile_options("/pathmap:${source_path}=${mapped}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      add_compile_options("-ffile-prefix-map=${source_path}=${mapped}")
    endif()
  endforeach()
endforeach()
add_subdirectory("${SHERPA_ONNX_SOURCE_DIR}" sherpa EXCLUDE_FROM_ALL)
add_executable(speech-input-cli speech-input-cli.cc)
target_link_libraries(speech-input-cli PRIVATE sherpa-onnx-c-api)
if(APPLE)
  target_link_options(speech-input-cli PRIVATE -Wl,-dead_strip)
elseif(MSVC)
  target_compile_options(speech-input-cli PRIVATE /utf-8)
  target_link_options(speech-input-cli PRIVATE /OPT:REF /OPT:ICF)
endif()
install(TARGETS speech-input-cli RUNTIME DESTINATION bin)
